"use client";

import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { supabase } from "../../lib/supabase";

const welcomeMessage = {
  role: "assistant",
  text: "Szia! 👋 A FlowPilot AI üzleti asszisztense vagyok. Miben segíthetek ma?"
};

export default function Dashboard() {
  const router = useRouter();

  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [sending, setSending] = useState(false);
  const [creating, setCreating] = useState(false);
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");
  const [remaining, setRemaining] = useState(null);

  const [conversations, setConversations] = useState([]);
  const [editingId, setEditingId] = useState(null);
const [editingTitle, setEditingTitle] = useState("");
const [deletingId, setDeletingId] = useState(null);
  const [activeId, setActiveId] = useState(null);
  const [messages, setMessages] = useState([welcomeMessage]);

  useEffect(() => {
    let active = true;

    async function loadDashboard() {
      const { data, error: authError } =
        await supabase.auth.getUser();

      if (!active) return;

      if (authError || !data.user) {
        router.replace("/login");
        return;
      }

      setUser(data.user);

      const { data: chats, error: chatError } =
        await supabase
          .from("conversations")
          .select("id, title, created_at")
          .eq("user_id", data.user.id)
          .order("created_at", { ascending: false });

      if (!active) return;

      if (chatError) {
        console.error("Beszélgetéslista hiba:", chatError);
        setError("Nem sikerült betölteni a beszélgetéseket.");
      } else {
        setConversations(chats || []);

        if (chats?.length) {
          setActiveId(chats[0].id);
        }
      }

      setLoading(false);
    }

    loadDashboard();

    return () => {
      active = false;
    };
  }, [router]);

  useEffect(() => {
    if (!activeId) {
      setMessages([welcomeMessage]);
      return;
    }

    let active = true;

    async function loadMessages() {
      const { data, error: historyError } =
        await supabase
          .from("chat_messages")
          .select("id, role, content, created_at")
          .eq("conversation_id", activeId)
          .order("created_at", { ascending: true })
          .order("id", { ascending: true })
          .limit(200);

      if (!active) return;

      if (historyError) {
        console.error("Üzenetbetöltési hiba:", historyError);
        setError("Nem sikerült betölteni az üzeneteket.");
      } else {
        setMessages(
          data?.length
            ? data.map((item) => ({
                id: item.id,
                role: item.role,
                text: item.content
              }))
            : [welcomeMessage]
        );
      }
    }

    loadMessages();

    return () => {
      active = false;
    };
  }, [activeId]);

  async function createConversation() {
    if (!user || creating || sending) return;

    setCreating(true);
    setError("");

    try {
      const { data, error: createError } =
        await supabase
          .from("conversations")
          .insert({
            user_id: user.id,
            title: "Új beszélgetés"
          })
          .select("id, title, created_at")
          .single();

      if (createError) throw createError;

      setConversations((previous) => [data, ...previous]);
      setActiveId(data.id);
      setMessages([welcomeMessage]);
      setMessage("");
    } catch (err) {
      console.error("Beszélgetés létrehozási hiba:", err);
      setError("Nem sikerült új beszélgetést létrehozni.");
    } finally {
      setCreating(false);
    }
  }

  async function selectConversation(id) {
    if (sending || creating || id === activeId) return;

    setError("");
    setMessage("");
    setActiveId(id);
  }

  async function sendMessage(e) {
    e.preventDefault();

    const text = message.trim();

    if (!text || sending || creating || !user ||
        text.length > 2000) return;

    if (!activeId) {
      setError("Először hozz létre egy új beszélgetést.");
      return;
    }

    const conversationId = activeId;

    setSending(true);
    setError("");

    try {
      const { data: saved, error: saveError } =
        await supabase
          .from("chat_messages")
          .insert({
            user_id: user.id,
            conversation_id: conversationId,
            role: "user",
            content: text
          })
          .select("id")
          .single();

      if (saveError) throw saveError;

      setMessages((previous) => [
        ...previous.filter((item) => item.id !== "welcome"),
        { id: saved.id, role: "user", text }
      ]);

      setMessage("");

      const { data: sessionData, error: sessionError } =
        await supabase.auth.getSession();

      if (sessionError || !sessionData.session?.access_token) {
        router.replace("/login");
        return;
      }

      const response = await fetch("/api/chat", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization:
            `Bearer ${sessionData.session.access_token}`
        },
        body: JSON.stringify({
          message: text,
          conversation_id: conversationId
        })
      });

      const result = await response.json();

      if (typeof result.remaining === "number") {
        setRemaining(result.remaining);
      }

      if (!response.ok) {
        throw new Error(
          result.error || "Nem sikerült választ kérni."
        );
      }

      setMessages((previous) => [
        ...previous,
        {
          role: "assistant",
          text: result.reply || "Nem érkezett válasz."
        }
      ]);

      // Az első üzenetből beszélgetéscím készítése
      const current = conversations.find(
        (chat) => chat.id === conversationId
      );

      if (current?.title === "Új beszélgetés") {
        const newTitle = text.slice(0, 55);

        const { error: titleError } = await supabase
          .from("conversations")
          .update({ title: newTitle })
          .eq("id", conversationId)
          .eq("user_id", user.id);

        if (!titleError) {
          setConversations((previous) =>
            previous.map((chat) =>
              chat.id === conversationId
                ? { ...chat, title: newTitle }
                : chat
            )
          );
        }
      }
    } catch (err) {
      console.error("Chat hiba:", err);
      setError(err.message || "Ismeretlen hiba történt.");
    } finally {
      setSending(false);
    }
  }
async function renameConversation(id) {
  if (!user || sending || creating) return;

  const title = editingTitle.trim();

  if (!title || title.length > 100) {
    setError("A cím 1–100 karakter hosszú lehet.");
    return;
  }

  setError("");

  const { data, error: renameError } = await supabase
    .from("conversations")
    .update({ title })
    .eq("id", id)
    .eq("user_id", user.id)
    .select("id")
    .single();

  if (renameError || !data) {
    console.error("Átnevezési hiba:", renameError);
    setError("Nem sikerült átnevezni a beszélgetést.");
    return;
  }

  setConversations((previous) =>
    previous.map((chat) =>
      chat.id === id ? { ...chat, title } : chat
    )
  );

  setEditingId(null);
  setEditingTitle("");
}

async function deleteConversation(id) {
  if (!user || sending || creating) return;

  setError("");

  const { data, error: deleteError } = await supabase
    .from("conversations")
    .delete()
    .eq("id", id)
    .eq("user_id", user.id)
    .select("id")
    .single();

  if (deleteError || !data) {
    console.error("Törlési hiba:", deleteError);
    setError("Nem sikerült törölni a beszélgetést.");
    return;
  }

  const updated = conversations.filter(
    (chat) => chat.id !== id
  );

  setConversations(updated);

  if (activeId === id) {
    setActiveId(updated[0]?.id || null);
    setMessages([welcomeMessage]);
  }

  setDeletingId(null);
  setEditingId(null);
}
  async function logout() {
    await supabase.auth.signOut();
    router.replace("/login");
  }

  const buttonStyle = {
    border: "none",
    borderRadius: 10,
    padding: "12px 16px",
    color: "white",
    cursor: "pointer",
    fontWeight: "bold"
  };

  if (loading) {
    return (
      <main style={{
        minHeight: "100vh",
        background: "#0b1120",
        color: "white",
        padding: 40
      }}>
        Fiók ellenőrzése...
      </main>
    );
  }

  return (
    <main style={{
      minHeight: "100vh",
      background: "#0b1120",
      color: "white",
      fontFamily: "Arial, sans-serif",
      padding: 24
    }}>
      <div style={{ maxWidth: 1200, margin: "0 auto" }}>
        <header style={{
          display: "flex",
          justifyContent: "space-between",
          alignItems: "center",
          gap: 16,
          flexWrap: "wrap",
          marginBottom: 24
        }}>
          <div>
            <h1 style={{ color: "#60a5fa", marginBottom: 4 }}>
              FlowPilot AI
            </h1>
            <p style={{ color: "#94a3b8" }}>
              Business Assistant
            </p>
          </div>

          <button
            onClick={logout}
            style={{ ...buttonStyle, background: "#dc2626" }}
          >
            Kijelentkezés
          </button>
        </header>

        <p style={{ color: "#cbd5e1" }}>
          Bejelentkezve: {user?.email}
        </p>

        <div style={{
          display: "grid",
          gridTemplateColumns:
            "repeat(auto-fit, minmax(min(100%, 280px), 1fr))",
          gap: 20,
          marginTop: 20
        }}>
          <aside style={{
            background: "#1e293b",
            padding: 20,
            borderRadius: 16
          }}>
            <button
              onClick={createConversation}
              disabled={creating || sending}
              style={{
                ...buttonStyle,
                width: "100%",
                background: "#2563eb"
              }}
            >
              {creating ? "Létrehozás..." : "+ Új beszélgetés"}
            </button>

            <h3 style={{ marginTop: 25 }}>
              Beszélgetések
            </h3>

            {conversations.length === 0 && (
              <p style={{ color: "#94a3b8" }}>
                Még nincs beszélgetésed.
              </p>
            )}

            <div style={{
              display: "flex",
              flexDirection: "column",
              gap: 8
            }}>
            

            {error && (
              <p role="alert" style={{ color: "#fca5a5" }}>
                {error}
              </p>
            )}
{conversations.map((chat) => (
  <div
    key={chat.id}
    style={{
      padding: 10,
      borderRadius: 10,
      background: activeId === chat.id
        ? "#1d4ed8"
        : "#0f172a",
      border: activeId === chat.id
        ? "1px solid #60a5fa"
        : "1px solid #475569",
      marginBottom: 8
    }}
  >
    {editingId === chat.id ? (
      <form
        onSubmit={(e) => {
          e.preventDefault();
          renameConversation(chat.id);
        }}
      >
        <input
          value={editingTitle}
          onChange={(e) =>
            setEditingTitle(e.target.value)
          }
          maxLength={100}
          autoFocus
          style={{
            width: "100%",
            padding: 9,
            boxSizing: "border-box",
            borderRadius: 7,
            border: "1px solid #64748b",
            background: "#1e293b",
            color: "white"
          }}
        />

        <div style={{
          display: "flex",
          gap: 8,
          marginTop: 8
        }}>
          <button type="submit">
            Mentés
          </button>

          <button
            type="button"
            onClick={() => setEditingId(null)}
          >
            Mégse
          </button>
        </div>
      </form>
    ) : deletingId === chat.id ? (
      <div>
        <p style={{ fontSize: 13 }}>
          Biztosan törlöd? Az üzenetek is törlődnek!
        </p>

        <div style={{ display: "flex", gap: 8 }}>
          <button
            type="button"
            onClick={() => deleteConversation(chat.id)}
            style={{ color: "#dc2626" }}
          >
            Igen, törlés
          </button>

          <button
            type="button"
            onClick={() => setDeletingId(null)}
          >
            Mégse
          </button>
        </div>
      </div>
    ) : (
      <div style={{
        display: "flex",
        alignItems: "center",
        gap: 8
      }}>
        <button
          type="button"
          onClick={() => selectConversation(chat.id)}
          disabled={sending || creating}
          style={{
            flex: 1,
            minWidth: 0,
            textAlign: "left",
            background: "transparent",
            border: "none",
            color: "white",
            cursor: "pointer",
            overflowWrap: "anywhere"
          }}
        >
          {chat.title}
        </button>

        <button
          type="button"
          title="Átnevezés"
          disabled={sending || creating}
          onClick={() => {
            setEditingId(chat.id);
            setEditingTitle(chat.title);
            setDeletingId(null);
          }}
        >
          ✏️
        </button>

        <button
          type="button"
          title="Törlés"
          disabled={sending || creating}
          onClick={() => {
            setDeletingId(chat.id);
            setEditingId(null);
          }}
        >
          🗑️
        </button>
      </div>
    )}
  </div>
))}
  </div>
        </aside>

        <section style={{
          background: "#1e293b",
          padding: 20,
          borderRadius: 16,
          minWidth: 0
        }}>
          <h2>
            {conversations.find(
              (chat) => chat.id === activeId
            )?.title || "Új beszélgetés"}
          </h2>

          <p style={{ color: "#93c5fd" }}>
            DEMO mód – tesztelés
          </p>

          <div
            role="log"
            aria-label="AI beszélgetés"
            style={{
              minHeight: 280,
              maxHeight: 450,
              overflowY: "auto",
              display: "flex",
              flexDirection: "column",
              gap: 14,
              padding: "20px 0"
            }}
          >
            {messages.map((item, index) => (
              <div
                key={item.id || index}
                style={{
                  alignSelf: item.role === "user"
                    ? "flex-end"
                    : "flex-start",
                  padding: 14,
                  borderRadius: 12,
                  background: item.role === "user"
                    ? "#2563eb"
                    : "#334155",
                  whiteSpace: "pre-wrap",
                  overflowWrap: "anywhere"
                }}
              >
                {item.text}
              </div>
            ))}

            {sending && <p>FlowPilot AI gondolkodik...</p>}
          </div>
            <form onSubmit={sendMessage}>
              <textarea
                value={message}
                onChange={(e) => setMessage(e.target.value)}
                placeholder="Írd ide a kérdésed..."
                maxLength={2000}
                rows={3}
                disabled={sending || !activeId}
                style={{
                  width: "100%",
                  padding: 14,
                  borderRadius: 10,
                  border: "1px solid #475569",
                  background: "#0f172a",
                  color: "white",
                  boxSizing: "border-box",
                  fontFamily: "inherit",
                  resize: "vertical"
                }}
              />

              <button
                type="submit"
                disabled={
                  sending || !activeId || !message.trim()
                }
                style={{
                  ...buttonStyle,
                  width: "100%",
                  marginTop: 12,
                  background: "#2563eb"
                }}
              >
                {sending
                  ? "Válasz készítése..."
                  : "Üzenet küldése ➤"}
              </button>
            </form>
          </section>
        </div>
      </div>
    </main>
  );
}
